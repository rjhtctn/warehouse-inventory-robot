cmake_minimum_required(VERSION 3.20)
project(warehouse_robot_sim)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(ros_gz_bridge REQUIRED)
find_package(rviz_common REQUIRED)
find_package(Qt6 REQUIRED COMPONENTS Widgets)
add_executable(simulation_bridge src/simulation_bridge.cpp)
target_compile_features(simulation_bridge PRIVATE cxx_std_17)
target_link_libraries(simulation_bridge rclcpp::rclcpp ros_gz_bridge::ros_gz_bridge)
install(TARGETS simulation_bridge DESTINATION lib/${PROJECT_NAME})
add_executable(robot_view src/robot_view.cpp)
target_compile_features(robot_view PRIVATE cxx_std_17)
target_link_libraries(robot_view rclcpp::rclcpp rviz_common::rviz_common Qt6::Widgets ${CMAKE_DL_LIBS})
install(TARGETS robot_view DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY launch config worlds meshes DESTINATION share/${PROJECT_NAME})
ament_package()
